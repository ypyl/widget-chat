var chartState = null;

function drawChart(multiplier, highlightPeak) {
  var series = [
    { label: 'Mon', value: 21 },
    { label: 'Tue', value: 27 },
    { label: 'Wed', value: 24 },
    { label: 'Thu', value: 38 },
    { label: 'Fri', value: 33 },
    { label: 'Sat', value: 14 },
    { label: 'Sun', value: 11 }
  ];

  var values = series.map(function (item) { return item.value * multiplier; });
  var max = Math.max.apply(null, values.concat([1]));
  var peakIndex = values.indexOf(Math.max.apply(null, values));

  var left = 26;
  var top = 12;
  var width = 360 - left - 10;
  var height = 160;
  var slot = width / series.length;
  var barWidth = Math.min(28, slot * 0.62);

  var barsHost = document.getElementById('bars');
  var labelHost = document.getElementById('labels');
  var gridHost = document.getElementById('grid');
  barsHost.textContent = '';
  labelHost.textContent = '';
  gridHost.textContent = '';

  var svgNs = 'http://www.w3.org/2000/svg';

  [0, 0.5, 1].forEach(function (fraction) {
    var line = document.createElementNS(svgNs, 'line');
    var y = top + height - height * fraction;
    line.setAttribute('x1', String(left - 6));
    line.setAttribute('x2', String(360 - 8));
    line.setAttribute('y1', String(y));
    line.setAttribute('y2', String(y));
    line.setAttribute('class', 'grid');
    gridHost.appendChild(line);

    var tick = document.createElementNS(svgNs, 'text');
    tick.setAttribute('x', String(left - 10));
    tick.setAttribute('y', String(y + 3));
    tick.setAttribute('text-anchor', 'end');
    tick.setAttribute('class', 'axis');
    tick.textContent = String(Math.round(max * fraction));
    gridHost.appendChild(tick);
  });

  series.forEach(function (item, index) {
    var barHeight = Math.max(2, (values[index] / max) * height);
    var x = left + slot * index + (slot - barWidth) / 2;

    var rect = document.createElementNS(svgNs, 'rect');
    rect.setAttribute('x', String(x));
    rect.setAttribute('y', String(top + height - barHeight));
    rect.setAttribute('width', String(barWidth));
    rect.setAttribute('height', String(barHeight));
    rect.setAttribute('rx', '3');
    rect.setAttribute('class', highlightPeak && index !== peakIndex ? 'bar dim' : 'bar');
    barsHost.appendChild(rect);

    var valueText = document.createElementNS(svgNs, 'text');
    valueText.setAttribute('x', String(x + barWidth / 2));
    valueText.setAttribute('y', String(top + height - barHeight - 4));
    valueText.setAttribute('text-anchor', 'middle');
    valueText.setAttribute('class', 'value');
    valueText.textContent = String(Math.round(values[index]));
    barsHost.appendChild(valueText);

    var name = document.createElementNS(svgNs, 'text');
    name.setAttribute('x', String(x + barWidth / 2));
    name.setAttribute('y', String(top + height + 14));
    name.setAttribute('text-anchor', 'middle');
    name.setAttribute('class', 'axis');
    name.textContent = item.label;
    labelHost.appendChild(name);
  });

  document.getElementById('summary').textContent =
    'Peak is ' + series[peakIndex].label + ' with ' + Math.round(values[peakIndex]) +
    ' signups at a ' + multiplier.toFixed(1) + ' times scale. Values are illustrative, not live data.';
}

function setupChart() {
  var slider = document.getElementById('scale');
  var output = document.getElementById('scalev');
  var peakButton = document.getElementById('peak');
  var highlightPeak = false;

  function refresh() {
    var multiplier = slider.valueAsNumber;
    if (!Number.isFinite(multiplier) || multiplier < 0.5 || multiplier > 2) {
      multiplier = 1;
    }
    output.textContent = multiplier.toFixed(1) + '\u00d7';
    drawChart(multiplier, highlightPeak);
  }

  slider.addEventListener('input', refresh);
  peakButton.addEventListener('click', function () {
    highlightPeak = !highlightPeak;
    peakButton.setAttribute('aria-pressed', highlightPeak ? 'true' : 'false');
    refresh();
  });

  chartState = { refresh: refresh };
  refresh();
}

setupChart();
